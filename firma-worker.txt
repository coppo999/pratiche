  if (!r || r.stato === "bozza" || r.tokHash !== await sha(tok)) throw new Err(404, "Link non valido o scaduto.");
  const blocco = () => { if (r.stato !== "attesa") throw new Err(409, "Questo documento è già stato firmato."); };
  if (az === "info") {
    if (r.stato === "attesa" && !r.aperto) { r.aperto = nowIso(); await putR(env, r); }
    const mail = !!(r.email && await env.KV.get("cfg:mailer"));
    return J({ stato: r.stato, doc: r.doc, nome: r.nome, studio: STUDIO, scade: r.exp, consenso: CONSENSO, privacy: PRIVACY, via: mail ? "email" : "studio", dest: mail ? mask(r.email) : "", finale: !!r.finale, ev: r.stato === "attesa" ? null : { firmatario: r.ev?.firmatario, at: r.ev?.firmaAt } });
  }
  if (az === "otp") {
    blocco();
    if (!r.email || !(await env.KV.get("cfg:mailer"))) return J({ via: "studio" });
    if (r.invii >= 6) throw new Err(429, "Troppi invii. Contatta lo studio per un nuovo codice.");
    if (r.otpAt && Date.now() - Date.parse(r.otpAt) < 45000) throw new Err(429, "Ti ho appena mandato un codice: controlla anche nello spam, oppure riprova tra un minuto.");
    const c = codice();
    r.otpHash = await sha(id + c); r.otpAt = nowIso(); r.otpExp = new Date(Date.now() + 15 * 60000).toISOString(); r.invii = (r.invii || 0) + 1; r.tent = 0;
    await putR(env, r);
    try { await mailer(env, { to: r.email, subject: `Codice per firmare: ${c}`, html: avvolgi("Il tuo codice di verifica", `<p>Per firmare «${escH(r.doc)}» inserisci questo codice:</p><p style="font-size:30px;font-weight:bold;letter-spacing:6px;margin:14px 0">${c}</p><p>Vale 15 minuti. Se non hai chiesto tu di firmare, ignora questa e-mail.</p>`) }); }
    catch (e) { console.error(e); return J({ via: "studio", errore: true }); }
    return J({ via: "email", dest: mask(r.email) });
  }
  if (az === "verifica") {
    blocco();
    const b = await leggiJ(req, 2000), cod = String(b.codice || "").replace(/\D/g, "");
    if ((r.tentTot || 0) >= 30 || (r.tent || 0) >= 5) throw new Err(429, "Troppi tentativi errati. " + (r.otpHash ? "Chiedi un nuovo codice." : "Contatta lo studio per un nuovo codice."));
    const h = await sha(id + cod);
    let metodo = null;
    if (r.otpHash && h === r.otpHash && Date.now() < Date.parse(r.otpExp)) metodo = "email";
    else if (r.codHash && h === r.codHash) metodo = "studio";
    if (!metodo) { r.tent = (r.tent || 0) + 1; r.tentTot = (r.tentTot || 0) + 1; await putR(env, r); throw new Err(401, "Codice errato o scaduto."); }
    const sess = rnd(24);
    r.sessHash = await sha(sess); r.sessExp = new Date(Date.now() + 2 * 3600000).toISOString(); r.metodo = metodo; r.verificatoAt = nowIso(); r.tent = 0; delete r.otpHash;
    await putR(env, r);
    return J({ sess, sha256: r.sha256, fields: r.fields, pagine: r.pagine });
  }
  if (az === "finale") {
    if (!r.finale) throw new Err(404, "La copia definitiva non è ancora pronta.");
    const f = await env.KV.get("fin:" + id, "arrayBuffer");
    if (!f) throw new Err(404, "La copia non è più disponibile: chiedila allo studio.");
    return new Response(f, { headers: { "content-type": "application/pdf", "content-disposition": `attachment; filename="documento_firmato.pdf"`, ...NOSTORE } });
  }
  const sessOk = async s => r.sessHash && s && r.sessHash === await sha(s) && Date.now() < Date.parse(r.sessExp);
  if (az === "pdf") {
    if (!await sessOk(req.headers.get("x-sess"))) throw new Err(401, "Sessione scaduta: ricarica la pagina.");
    const f = await env.KV.get("pdf:" + id, "arrayBuffer");
    if (!f) throw new Err(404, "Documento non disponibile.");
    return new Response(f, { headers: { "content-type": "application/pdf", ...NOSTORE } });
  }
  if (az === "firma") {
    blocco();
    const b = await leggiJ(req);
    if (!await sessOk(b.sess)) throw new Err(401, "Sessione scaduta: ricarica la pagina.");
    const nome = String(b.nome || "").trim().slice(0, 120);
    if (nome.length < 3) throw new Err(400, "Scrivi nome e cognome.");
    if (b.consenso !== true) throw new Err(400, "Serve la conferma di aver letto il documento.");
    const m = /^data:image\/png;base64,([A-Za-z0-9+/=]+)$/.exec(String(b.png || ""));
    if (!m || m[1].length > 500000 || !m[1].startsWith("iVBORw0KGgo")) throw new Err(400, "Firma non valida: riprova.");
    const cf = req.cf || {};
    r.ev = {
      id, documento: r.doc, sha256Originale: r.sha256, pagine: r.pagine, nomeIndicato: r.nome, firmatario: nome,
      verifica: r.metodo === "email" ? `Codice monouso inviato a ${r.email}` : "Codice consegnato dallo studio con messaggio separato",
      creatoAt: r.creato, apertoAt: r.aperto || "", verificatoAt: r.verificatoAt, firmaAt: nowIso(),
      ip: req.headers.get("cf-connecting-ip") || "", paese: cf.country || "", dispositivo: String(req.headers.get("user-agent") || "").slice(0, 300), consenso: CONSENSO, studio: `${STUDIO.nome}, ${STUDIO.indirizzo}`,
    };
    r.stato = "firmato"; delete r.sessHash;
    await env.KV.put("sig:" + id, m[1], { expirationTtl: ttlDi(r) });
    await putR(env, r);
    ctx.waitUntil(mailer(env, { to: STUDIO.email, subject: `Firmato: ${r.doc} – ${nome}`, html: avvolgi("Documento firmato", `<p><b>${escH(nome)}</b> ha firmato «${escH(r.doc)}» il ${escH(dataIt(r.ev.firmaAt))}.</p><p>La copia firmata entra nella pratica ${escH(r.anno)}/${escH(r.dir)} appena Pratiche Studio è aperto.</p>`) }).catch(e => console.error(e)));
    return J({ ok: true, ev: r.ev });
  }
  throw new Err(404, "Operazione non valida.");
}

/* ---------- Area riservata: Pratiche Studio (PC e telefono) ---------- */
async function apiStudio(req, env, ctx, parts, url) {
  const [az, id] = parts;
  if (!env.APP_KEY || req.headers.get("x-key") !== env.APP_KEY) throw new Err(401, "Chiave del servizio firma errata.");
  if (az === "ping") return J({ ok: true, mailer: !!(await env.KV.get("cfg:mailer")) }, 200, true);
  if (az === "config") {
    const b = await leggiJ(req, 3000);
    if (!/^https:\/\/script\.google\.com\/macros\/s\/[\w-]+\/exec$/.test(b.url || "") || !b.key) throw new Err(400, "L'indirizzo deve essere quello dell'app web di Google Apps Script (termina con /exec).");
    await env.KV.put("cfg:mailer", JSON.stringify({ url: b.url, key: b.key }));
    return J({ ok: true }, 200, true);
  }
  if (az === "nuova") {
    const b = await leggiJ(req, 60000);
    if (!b.doc || !/^[0-9a-f]{64}$/.test(b.sha256 || "") || !Array.isArray(b.fields) || !b.fields.length) throw new Err(400, "Dati incompleti.");
    if (b.email && !emailOk(b.email)) throw new Err(400, "Indirizzo e-mail non valido.");
    const giorni = Math.min(60, Math.max(1, +b.giorni || 30));
    const nid = rnd(8), tok = rnd(24), cod = codice();
    const r = { id: nid, stato: "bozza", doc: String(b.doc).slice(0, 200), nome: String(b.nome || "").slice(0, 120), email: b.email || "", tel: String(b.tel || "").slice(0, 40), anno: b.anno, dir: b.dir, path: b.path || [], fields: b.fields, sha256: b.sha256, pagine: +b.pagine || 0, creato: nowIso(), exp: new Date(Date.now() + giorni * 86400000).toISOString(), tokHash: await sha(tok), codHash: await sha(nid + cod) };
    await putR(env, r);
    return J({ id: nid, link: `${url.origin}/s/${nid}/${tok}`, codice: cod, scade: r.exp }, 200, true);
  }
  const r = await getR(env, id);
  if (!r) throw new Err(404, "Richiesta non trovata o scaduta.");
  if (az === "pdf" && req.method === "PUT") {
    const buf = await req.arrayBuffer();
    if (buf.byteLength > 24e6) throw new Err(413, "PDF troppo grande (massimo 24 MB).");
    await env.KV.put("pdf:" + id, buf, { expirationTtl: ttlDi(r) });
    r.stato = "attesa"; await putR(env, r);
    const idx = await getIdx(env); if (!idx.includes(id)) { idx.push(id); await env.KV.put("index", JSON.stringify(idx)); }
    return J({ ok: true }, 200, true);
  }
  if (az === "pdf") { const f = await env.KV.get("pdf:" + id, "arrayBuffer"); if (!f) throw new Err(404, "Originale non più disponibile."); return new Response(f, { headers: { "content-type": "application/pdf", ...NOSTORE, ...CORS } }); }
  if (az === "dati") return J({ ...pub(r), fields: r.fields, png: await env.KV.get("sig:" + id), ev: r.ev }, 200, true);
  if (az === "codice") { const c = codice(); r.codHash = await sha(id + c); r.tent = 0; await putR(env, r); return J({ codice: c }, 200, true); }
  if (az === "presa") {
    const b = await leggiJ(req, 500);
    if (r.presa && r.presa.by !== b.by && Date.now() - Date.parse(r.presa.at) < 300000) return J({ ok: false }, 200, true);
    r.presa = { by: String(b.by || "").slice(0, 40), at: nowIso() }; await putR(env, r);
    return J({ ok: true }, 200, true);
  }
  if (az === "finale") {
    const buf = await req.arrayBuffer();
    if (buf.byteLength > 24.5e6) throw new Err(413, "PDF troppo grande.");
    r.exp = new Date(Date.now() + 30 * 86400000).toISOString(); r.finale = true; r.stato = "archiviato";
    const tok = url.searchParams.get("t") || "";
    await env.KV.put("fin:" + id, buf, { expirationTtl: ttlDi(r) });
    await putR(env, r);
    await env.KV.delete("pdf:" + id); await env.KV.delete("sig:" + id);
    await env.KV.put("index", JSON.stringify((await getIdx(env)).filter(x => x !== id)));
    if (r.email && tok && r.tokHash === await sha(tok)) ctx.waitUntil(mailer(env, { to: r.email, subject: `Copia firmata: ${r.doc}`, html: avvolgi("Documento firmato", `<p>In allegato la copia firmata di «${escH(r.doc)}», con il registro della firma nell'ultima pagina. Conservala: da questo servizio viene cancellata dopo 30 giorni.</p>`), attachUrl: `${url.origin}/api/${id}/${tok}/finale`, attachName: r.doc.replace(/\.pdf$/i, "") + "_firmato.pdf" }).catch(e => console.error(e)));
    return J({ ok: true }, 200, true);
  }
  if (az === "invia") {
    const b = await leggiJ(req, 6000);
    if (!emailOk(b.to)) throw new Err(400, "Indirizzo e-mail non valido.");
    const html = avvolgi(b.titolo || "Documento da firmare", String(b.testo || "").split(/\n{2,}/).map(p => `<p>${escH(p).replace(/\n/g, "<br>").replace(/(https:\/\/[^\s<]+)/g, '<a href="$1">$1</a>')}</p>`).join(""));
    if (!await mailer(env, { to: b.to, subject: String(b.oggetto || "Documento da firmare").slice(0, 150), html })) return J({ ok: false }, 200, true);
    return J({ ok: true }, 200, true);
  }
  if (az === "annulla") {
    await Promise.all(["req:", "pdf:", "sig:", "fin:"].map(k => env.KV.delete(k + id)));
    await env.KV.put("index", JSON.stringify((await getIdx(env)).filter(x => x !== id)));
    return J({ ok: true }, 200, true);
  }
  throw new Err(404, "Operazione non valida.");
}
const pub = r => ({ id: r.id, stato: r.stato, doc: r.doc, nome: r.nome, anno: r.anno, dir: r.dir, path: r.path, aperto: r.aperto || "", firmaAt: r.ev?.firmaAt || "", firmatario: r.ev?.firmatario || "", exp: r.exp, invii: r.invii || 0 });

export default {
  async fetch(req, env, ctx) {
    const url = new URL(req.url), seg = url.pathname.split("/").filter(Boolean);
    try {
      if (req.method === "OPTIONS") return new Response(null, { headers: CORS });
      if (seg[0] === "admin") {
        if (seg[1] === "attive") {
          if (!env.APP_KEY || req.headers.get("x-key") !== env.APP_KEY) throw new Err(401, "Chiave del servizio firma errata.");
          const idx = await getIdx(env), rs = await Promise.all(idx.map(i => getR(env, i)));
          const buoni = rs.filter(Boolean);
          if (buoni.length !== idx.length) await env.KV.put("index", JSON.stringify(buoni.map(r => r.id)));
          return J({ items: buoni.map(pub) }, 200, true);
        }
        return await apiStudio(req, env, ctx, seg.slice(1), url);
      }
      if (seg[0] === "api" && seg.length === 4) return await apiCliente(req, env, ctx, seg[1], seg[2], seg[3]);
      if (seg[0] === "kit.js") return new Response(ASSETS.kit, { headers: { "content-type": "text/javascript; charset=utf-8", "cache-control": "no-cache", ...CORS } });
      if (seg[0] === "firma.js") return new Response(ASSETS.firma, { headers: { "content-type": "text/javascript; charset=utf-8", "cache-control": "no-cache" } });
      if (seg[0] === "s" && seg.length === 3) return new Response(ASSETS.pagina, { headers: { "content-type": "text/html; charset=utf-8", ...NOSTORE, "x-frame-options": "DENY", "content-security-policy": "default-src 'none'; script-src 'self' https://cdn.jsdelivr.net blob:; worker-src blob:; connect-src 'self' https://cdn.jsdelivr.net; style-src 'unsafe-inline'; img-src data: blob:; font-src 'none'; base-uri 'none'; form-action 'none'; frame-ancestors 'none'" } });
      if (seg.length === 0) return new Response("Servizio di firma dello " + STUDIO.nome, { headers: { "content-type": "text/plain; charset=utf-8" } });
      return new Response("Non trovato", { status: 404 });
    } catch (e) {
      if (!(e instanceof Err)) console.error(e);
      return J({ error: e instanceof Err ? e.message : "Errore del servizio, riprova." }, e.status || 500, true);
    }
  },
};
